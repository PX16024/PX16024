Plane Shapes Area Calculator

A program that fetches options and accepts user inputs to calculate a specific plane's area. 


#include <iostream>
#include <cmath>
using namespace std;

//Option [1]
void AS() {
    double m;
    cout << "----------------------------------------" << endl
         << "AREA OF SQUARE" << endl
         << "----------------------------------------" << endl
         << "Enter the measurement of one side of the square: ";
    cin >> m;
    cout << "The area of the square is " << pow(m, 2) << endl;
}

//Option [2]
void AR() {
    double len, wid;
    cout << "----------------------------------------" << endl
         << "AREA OF RECTANGLE" << endl
         << "----------------------------------------"<< endl
         << "Enter the length of the rectangle: ";
    cin >> len;
    cout << "Enter the width: ";
    cin >> wid;
    cout << "The area of the rectangle is " << len * wid << endl;
}

//Option [3]
void AT() {
    double blen, height;
    cout << "----------------------------------------" << endl
         << "AREA OF TRIANGLE" << endl
         << "----------------------------------------" << endl
         << "Enter the base length of the triangle: ";
    cin >> blen;
    cout << "Enter the height perpendicular to the base of the triangle: ";
    cin >> height;
    cout << "The area of the triangle is " << 0.5 * blen * height << endl;
}

//Option [4]
void AC() {
    double rad;
    cout << "----------------------------------------" << endl
         << "AREA OF CIRCLE" << endl
         << "----------------------------------------" << endl
         << "Enter the radius of the circle: ";
    cin >> rad;
    cout << "The area of the circle is " << M_PI * pow(rad, 2) << endl;
}

int main() {
    int choice;

    // do…while loop ensures menu runs at least once
    do {
        cout << "----------------------------------------" << endl
             << "MENU" << endl
             << "----------------------------------------" << endl
             << "[1] - Area of square" << endl
             << "[2] - Area of rectangle" << endl
             << "[3] - Area of triangle" << endl
             << "[4] - Area of circle" << endl
             << "[5] - Exit" << endl
             << "----------------------------------------" << endl
             << "Enter your choice: ";
        cin >> choice;

        switch(choice) {
            case 1: AS(); break;
            case 2: AR(); break;
            case 3: AT(); break;
            case 4: AC(); break;
            case 5: cout << "Aight cool, goodbye!" << endl; break;
            default: cout << "Invalid choice, try again!" << endl;
        }

    } while (choice != 5); // loop until Exit is chosen

    return 0;
}
